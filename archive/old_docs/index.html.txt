<!-- ═══════════════════════════════════════════════════════
     MEASUREMENT ENGINE  (Snap to Point + Line)
═══════════════════════════════════════════════════════ -->
<script>
const MeasurementEngine = (function () {
  const SETBACK_GUIDE_COLORS = ['#e91e63','#9c27b0','#ff5722'];
  const lawColors = {
    site:'rgba(0,86,179,0.4)', building:'rgba(220,53,69,0.4)', space:'rgba(40,167,69,0.4)',
    floor:'rgba(255,193,7,0.4)', parking:'rgba(255,120,0,0.4)',
    stair:'rgba(156,39,176,0.4)', roof:'rgba(150,150,150,0.4)',
  };
  const strokeColors = {
    site:'#0056b3', building:'#dc3545', space:'#28a745',
    floor:'#e6a800', parking:'#ff7800', stair:'#9c27b0', roof:'#666',
  };
  function _polyColor(t){ return lawColors[t]||'rgba(100,100,100,0.35)'; }
  function _strokeColor(t){ return strokeColors[t]||'#555'; }
  let SNAP_PT_THRESHOLD = 22, SNAP_LINE_THRESHOLD = 14;

  let _canvas=null,_getZoom=null,_onCalibrated=null,_onMeasured=null;
  let PIXELS_PER_METER=0;
  let tempPoints=[],tempLines=[],tempCircles=[],tempLabels=[];
  let _ghostLine=null,_snapIndicator=null;

  function init(c,gz,cb){ _canvas=c;_getZoom=gz;_onCalibrated=cb&&cb.onCalibrated||function(){};_onMeasured=cb&&cb.onMeasured||function(){}; }
  function resetCalibration(){ PIXELS_PER_METER=0; }
  function getPixelsPerMeter(){ return PIXELS_PER_METER; }
  function isCalibrated(){ return PIXELS_PER_METER>0; }
  function getTempPointCount(){ return tempPoints.length; }
  function sw(b){ return b/_getZoom(); }

  function _addDot(x,y,color,label){
    const r=sw(5);
    const c=new fabric.Circle({left:x-r,top:y-r,radius:r,fill:color,stroke:'#fff',strokeWidth:sw(1.5),selectable:false,evented:false});
    _canvas.add(c); tempCircles.push(c);
    if(label){ const t=new fabric.Text(label,{left:x+sw(8),top:y-sw(12),fontSize:Math.max(10,sw(14)),fill:'#fff',backgroundColor:color,padding:3,selectable:false,evented:false}); _canvas.add(t); tempLabels.push(t); }
  }
  function _addGuideLine(x1,y1,x2,y2,color,dashed){
    const l=new fabric.Line([x1,y1,x2,y2],{stroke:color,strokeWidth:sw(2),strokeDashArray:dashed?[sw(6),sw(4)]:null,selectable:false,evented:false});
    _canvas.add(l); tempLines.push(l); return l;
  }
  function _getAnchors(obj){
    if(!obj||!obj.customData) return [];
    if(obj.type==='polygon'&&obj.points) return obj.points;
    if(obj.type==='line') return [{x:obj.x1,y:obj.y1},{x:obj.x2,y:obj.y2}];
    return [];
  }
  // ── Snap flags (controlled by toolbar) ──
  const snapFlags = { vector: true, object: true, midpoint: true, ortho: true };
  function setSnapFlag(key, val){ snapFlags[key] = val; }
  function getSnapFlags(){ return snapFlags; }
  function setSnapThreshold(pt, ln){ SNAP_PT_THRESHOLD = pt; SNAP_LINE_THRESHOLD = ln; }
  function getSnapThreshold(){ return { pt: SNAP_PT_THRESHOLD, ln: SNAP_LINE_THRESHOLD }; }

  function snapPoint(pt, prevPt){
    // ── Close polygon: firstPt มี priority สูงสุด ──
    if(tempPoints.length >= 2){
      const first = tempPoints[0];
      const closeTol = SNAP_PT_THRESHOLD / _getZoom();
      if(Math.hypot(pt.x - first.x, pt.y - first.y) < closeTol){
        return {x: first.x, y: first.y, type: 'close'};
      }
    }
    // ── Orthogonal lock ──
    let constrained = {x: pt.x, y: pt.y};
    if(snapFlags.ortho && prevPt){
      const dx = Math.abs(pt.x - prevPt.x), dy = Math.abs(pt.y - prevPt.y);
      if(dx > dy) constrained = {x: pt.x, y: prevPt.y};
      else        constrained = {x: prevPt.x, y: pt.y};
    }
    const cp = constrained;

    const ptT = SNAP_PT_THRESHOLD/_getZoom(), lnT = SNAP_LINE_THRESHOLD/_getZoom();
    let nearest = null, minDist = ptT;

    // ── Vector endpoint snap ──
    if(snapFlags.vector && typeof VectorEngine !== 'undefined' && typeof pageNum !== 'undefined'){
      const vcands = VectorEngine.getCandidates(pageNum);
      if(vcands.length > 0){
        const vTol = Math.max(ptT, 10/_getZoom());
        let vNearest = null, vMin = vTol;
        for(let vi = 0; vi < vcands.length; vi++){
          const d = Math.hypot(cp.x - vcands[vi].x, cp.y - vcands[vi].y);
          if(d < vMin){ vMin = d; vNearest = vcands[vi]; }
        }
        if(vNearest) return {x: vNearest.x, y: vNearest.y, type: 'point'};
      }
    }

    // ── Object snap: endpoint ──
    if(snapFlags.object){
      const anchors = [];
      _canvas.getObjects().forEach(o => _getAnchors(o).forEach(a => anchors.push({x:a.x, y:a.y})));
      tempPoints.forEach(a => anchors.push({x:a.x, y:a.y}));
      anchors.forEach(a => {
        const d = Math.hypot(cp.x-a.x, cp.y-a.y);
        if(d < minDist){ minDist=d; nearest={x:a.x, y:a.y, type:'point'}; }
      });
      if(nearest) return nearest;
    }

    // ── Midpoint snap ──
    if(snapFlags.midpoint){
      minDist = ptT;
      _canvas.getObjects().forEach(obj => {
        let segs = [];
        if(obj.type==='line') segs=[{x1:obj.x1,y1:obj.y1,x2:obj.x2,y2:obj.y2}];
        else if(obj.type==='polygon'&&obj.points&&obj.points.length>1){
          for(let i=0;i<obj.points.length;i++){ const j=(i+1)%obj.points.length; segs.push({x1:obj.points[i].x,y1:obj.points[i].y,x2:obj.points[j].x,y2:obj.points[j].y}); }
        }
        segs.forEach(seg => {
          const mx = (seg.x1+seg.x2)/2, my = (seg.y1+seg.y2)/2;
          const d = Math.hypot(cp.x-mx, cp.y-my);
          if(d < minDist){ minDist=d; nearest={x:mx, y:my, type:'midpoint'}; }
        });
      });
      if(nearest) return nearest;
    }

    // ── Line (perpendicular foot) snap ──
    if(snapFlags.object){
      minDist = lnT;
      _canvas.getObjects().forEach(obj => {
        let segs = [];
        if(obj.type==='line') segs=[{x1:obj.x1,y1:obj.y1,x2:obj.x2,y2:obj.y2}];
        else if(obj.type==='polygon'&&obj.points&&obj.points.length>1){
          for(let i=0;i<obj.points.length;i++){ const j=(i+1)%obj.points.length; segs.push({x1:obj.points[i].x,y1:obj.points[i].y,x2:obj.points[j].x,y2:obj.points[j].y}); }
        }
        segs.forEach(seg => {
          const dx=seg.x2-seg.x1, dy=seg.y2-seg.y1, lenSq=dx*dx+dy*dy;
          if(!lenSq) return;
          const t=Math.max(0,Math.min(1,((cp.x-seg.x1)*dx+(cp.y-seg.y1)*dy)/lenSq));
          const foot={x:seg.x1+t*dx, y:seg.y1+t*dy};
          const d=Math.hypot(cp.x-foot.x, cp.y-foot.y);
          if(d < minDist){ minDist=d; nearest={x:foot.x, y:foot.y, type:'line'}; }
        });
      });
    }

    // ถ้าไม่ snap → return constrained point (ortho ยังมีผล)
    return nearest || {x: cp.x, y: cp.y, type: snapFlags.ortho && prevPt ? 'ortho' : null};

<!-- ═══════════════════════════════════════════════════════
     VECTOR ENGINE  (PDF type detection + snap candidates)
     ── IIFE แยกอิสระ ไม่แตะ MeasurementEngine / main app ──
═══════════════════════════════════════════════════════ -->
<script>
const VectorEngine = (function () {
  // ─── PDF.js OPS constants ────────────────────────────
  const OPS_MOVE_TO  = 13;
  const OPS_LINE_TO  = 14;
  const OPS_CURVE_TO = 15;

  // ─── Config ──────────────────────────────────────────
  const MAX_CANDS    = 1500;  // จุด snap สูงสุดต่อหน้า
  const CHUNK_SIZE   = 800;   // ops ต่อ chunk — yield ทุกๆ นี้เพื่อไม่ block UI
  const YIELD_MS     = 0;     // setTimeout(0) = yield กลับ event loop

  // ─── State ───────────────────────────────────────────
  const _candidates  = {};   // { pageNum: [{x,y},...] }
  const _typeCache   = {};   // { pageNum: 'vector'|'raster' }
  const _extracting  = {};   // { pageNum: Promise } — ป้องกัน extract ซ้ำ

  // ─── detectPDFType ───────────────────────────────────
  // scan แค่ 200 ops แรก — เจอ path op = vector, ไม่เจอ = raster
  async function detectPDFType(page) {
    const pn = page.pageNumber;
    if (_typeCache[pn] !== undefined) return _typeCache[pn];
    try {
      const opList = await page.getOperatorList();
      const fn = opList.fnArray;
      const scanDepth = Math.min(fn.length, 200); // cap ไว้แค่ 200 ops แรก
      let type = 'raster';
      for (let i = 0; i < scanDepth; i++) {
        const op = fn[i];
        if (op === OPS_MOVE_TO || op === OPS_LINE_TO || op === OPS_CURVE_TO) {
          type = 'vector'; break;
        }
      }
      _typeCache[pn] = type;
      return type;
    } catch (e) {
      _typeCache[pn] = 'raster';
      return 'raster';
    }
  }

  // ─── extractSnapCandidates ───────────────────────────
  // ON-DEMAND เท่านั้น: เรียกเฉพาะเมื่อ user กดปุ่ม measure
  // chunked processing: yield ทุก CHUNK_SIZE ops ป้องกัน hang
  function extractSnapCandidates(page, rotation) {
    const pn = page.pageNumber;
    if (_candidates[pn]) return Promise.resolve(_candidates[pn]);
    if (_extracting[pn]) return _extracting[pn]; // กำลัง extract อยู่แล้ว รอ promise เดิม

    const promise = _doExtract(page, rotation).then(result => {
      delete _extracting[pn];
      return result;
    });
    _extracting[pn] = promise;
    return promise;
  }

  async function _doExtract(page, rotation) {
    const pn  = page.pageNumber;
    const rot = rotation || 0;
    const vp  = page.getViewport({ scale: 1.5, rotation: rot });
    const W = vp.width, H = vp.height;
    const margin = 20;
    const [ta, tb, tc, td, te, tf] = vp.transform;

    function toCanvas(px, py) {
      return { x: ta*px + tc*py + te, y: tb*px + td*py + tf };
    }

    try {
      const opList = await page.getOperatorList();
      const fn  = opList.fnArray;
      const ar  = opList.argsArray;
      const total = fn.length;
      const pts = [];

      // วน loop แบบ chunked — yield ทุก CHUNK_SIZE ops
      for (let i = 0; i < total && pts.length < MAX_CANDS; i++) {
        // yield กลับ event loop ทุก chunk เพื่อไม่ block UI
        if (i > 0 && i % CHUNK_SIZE === 0) {
          await new Promise(r => setTimeout(r, YIELD_MS));
        }
        const op = fn[i];
        const a  = ar[i];
        if (!a) continue;
        if (op === OPS_MOVE_TO || op === OPS_LINE_TO) {
          if (a.length >= 2) {
            const p = toCanvas(a[0], a[1]);
            if (p.x > -margin && p.x < W+margin && p.y > -margin && p.y < H+margin)
              pts.push(p);
          }
        } else if (op === OPS_CURVE_TO) {
          if (a.length >= 6) {
            const p = toCanvas(a[4], a[5]);
            if (p.x > -margin && p.x < W+margin && p.y > -margin && p.y < H+margin)
              pts.push(p);
          }
        }
      }

      _candidates[pn] = pts;
      console.log(`[VectorEngine] หน้า ${pn}: extract ${pts.length} pts จาก ${total} ops`);
      return pts;
    } catch (e) {
      console.warn('[VectorEngine] extract error:', e);
      _candidates[pn] = [];
      return [];
    }
  }

  // ─── getCandidates ───────────────────────────────────
  // return ทันที (synchronous) — ใช้ใน snapPoint
  function getCandidates(pn) {
    return _candidates[pn] || [];
  }

  function getType(pn) {
    return _typeCache[pn] || null;
  }

  function isExtracting(pn) {
    return !!_extracting[pn];
  }

  function clearPage(pn) {
    delete _candidates[pn];
    delete _typeCache[pn];
    delete _extracting[pn];
  }

  function clearAll() {
    [_candidates, _typeCache, _extracting].forEach(obj =>
      Object.keys(obj).forEach(k => delete obj[k])
    );
  }

  return { detectPDFType, extractSnapCandidates, getCandidates, getType, isExtracting, clearPage, clearAll };
})();
</script>

// ─── สร้าง DOM ───────────────────────────────────
function buildLawPanel() {
  const container = document.getElementById('law-list');
  if (!container) return;
  container.innerHTML = '';
  LAW_DATA.forEach((grp, gi) => {
    const groupEl = document.createElement('div');
    groupEl.className = 'law-group';
    groupEl.id = `law-grp-${gi}`;

    const header = document.createElement('div');
    header.className = 'law-group-header';
    header.innerHTML = `<span>${grp.group}</span><span class="law-arrow">▼</span>`;
    header.onclick = () => groupEl.classList.toggle('open');

    const body = document.createElement('div');
    body.className = 'law-group-body';

    grp.items.forEach((item, ii) => {
      const el = document.createElement('div');
      el.className = 'law-item';
      el.draggable = true;
      el.dataset.lawNo = item.no;
      el.dataset.lawText = item.text;
      el.innerHTML = `<span class="law-no">${item.no}</span>${item.text}`;

      // ── Desktop: HTML5 drag ──
      el.addEventListener('dragstart', e => {
        el.classList.add('dragging');
        e.dataTransfer.setData('text/plain', JSON.stringify({ no: item.no, text: item.text }));
        e.dataTransfer.effectAllowed = 'copy';
      });
      el.addEventListener('dragend', () => el.classList.remove('dragging'));

      body.appendChild(el);
    });

    groupEl.appendChild(header);
    groupEl.appendChild(body);
    container.appendChild(groupEl);
  });
}

// ─── Filter ──────────────────────────────────────
function filterLawItems(q) {
  const kw = q.trim().toLowerCase();
  document.querySelectorAll('.law-group').forEach(grp => {
    let anyVisible = false;
    grp.querySelectorAll('.law-item').forEach(el => {
      const match = !kw || el.textContent.toLowerCase().includes(kw);
      el.style.display = match ? '' : 'none';
      if (match) anyVisible = true;
    });
    grp.style.display = anyVisible ? '' : 'none';
    if (kw && anyVisible) grp.classList.add('open');
    else if (!kw) grp.classList.remove('open');
  });
}

// ─── Drop zone บน canvas-scroll ─────────────────
function initLawDrop() {
  const scroll = document.getElementById('canvas-scroll');
  if (!scroll) return;

  scroll.addEventListener('dragover', e => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'copy';
  });

  scroll.addEventListener('drop', e => {
    e.preventDefault();
    if (!window.canvas || !window.pdfDoc) return;

    let data;
    try { data = JSON.parse(e.dataTransfer.getData('text/plain')); }
    catch { return; }

    // คำนวณพิกัดบน canvas (หัก offset + zoom)
    const rect = scroll.getBoundingClientRect();
    const rawX = e.clientX - rect.left;
    const rawY = e.clientY - rect.top;
    const cx = (rawX - (window.canvasOffsetX || 0)) / (window.currentZoom || 1);
    const cy = (rawY - (window.canvasOffsetY || 0)) / (window.currentZoom || 1);

    _dropLawText(data.no, data.text, cx, cy);
  });
}

// ─── วาง law text บน Fabric canvas ─────────────
function _dropLawText(no, text, cx, cy) {
  const color = document.getElementById('law-color')?.value || '#ff453a';
  const zoom  = window.currentZoom || 1;
  const fs    = Math.max(10, 13 / zoom);
  const pad   = 8 / zoom;
  const sw    = 1.5 / zoom;

  const fullText = `${no}\n${text}`;

  // IText ให้แก้ไขได้ + resize + rotate
  const txt = new fabric.IText(fullText, {
    left: cx, top: cy,
    fontSize: fs,
    fill: '#fff',
    fontFamily: '-apple-system, "Helvetica Neue", sans-serif',
    lineHeight: 1.4,
    editable: true,
    selectable: true,
    hasControls: true,
    hasBorders: true,
    lockScalingFlip: true,
    customData: { isLawText: true, lawNo: no },
    // กรอบ background ผ่าน backgroundColor
    backgroundColor: color + 'cc',
    padding: pad,
  });

  // เพิ่ม border สี
  txt.set({
    borderColor: color,
    cornerColor: color,
    cornerSize: 8 / zoom,
  });

  window.canvas.add(txt);
  window.canvas.setActiveObject(txt);
  window.canvas.renderAll();

  if (typeof saveCurrentPage === 'function') saveCurrentPage();
  if (typeof setMode === 'function') setMode('select');
}

// ─── init ────────────────────────────────────────
window.addEventListener('DOMContentLoaded', () => {
  buildLawPanel();
  initLawDrop();
});

// PDF LOAD  — render 10 หน้าแรก → เปิดใช้งาน
//             → render ที่เหลือ background เงียบๆ
// ──────────────────────────────────────────────
const pageImageCache = {};
const PRIORITY_COUNT = 10;
let _bgRenderActive = false;

function showLoading(){ document.getElementById('loading-overlay').classList.remove('hidden'); }
function hideLoading(){
  const el=document.getElementById('loading-overlay');
  el.classList.add('fade-out');
  setTimeout(()=>el.classList.add('hidden'),450);
}
function setLoadingProgress(done, total, msg){
  const pct=Math.round(done/total*100);
  document.getElementById('loading-bar').style.width=pct+'%';
  document.getElementById('loading-pct').textContent=pct+'%';
  document.getElementById('loading-sub').textContent=msg||`render หน้า ${done}/${total}...`;
}

async function renderPageToCache(pdf, i, srcId=0){
  const cacheKey = `${srcId}_${i}`;
  const page=await pdf.getPage(i);
  const rot=(page.rotate+(pageRotations[i]||0))%360;
  const vp=page.getViewport({scale:1.5, rotation:rot});
  const tmp=document.createElement('canvas');
  tmp.width=Math.round(vp.width); tmp.height=Math.round(vp.height);
  await page.render({canvasContext:tmp.getContext('2d',{alpha:false}), viewport:vp}).promise;
  pageImageCache[cacheKey]=tmp.toDataURL('image/jpeg',0.92);
  // thumbnail
  const vpT=page.getViewport({scale:0.25, rotation:rot});
  const tc=document.getElementById(`thumb-canvas-${cacheKey}`);
  if(tc){ tc.width=vpT.width; tc.height=vpT.height; await page.render({canvasContext:tc.getContext('2d'),viewport:vpT}).promise; }
}

async function renderBackground(pdf, startFrom){
  _bgRenderActive=true;
  const total=pdf.numPages;
  for(let i=startFrom;i<=total;i++){
    if(!_bgRenderActive) break;
    const cacheKey = `0_${i}`;
    if(!pageImageCache[cacheKey]){
      await renderPageToCache(pdf, i, 0);
      const th=document.getElementById(`thumb-0_${i}`);
      if(th) th.classList.remove('thumb-loading');
    }
  }
  _bgRenderActive=false;
}

document.getElementById('pdf-upload').addEventListener('change', function(e){
  const file=e.target.files[0];
  if(!file||file.type!=='application/pdf') return alert('กรุณาอัพโหลดไฟล์ PDF');
  _bgRenderActive=false;
  pdfFileName=file.name.replace(/\.pdf$/i,'');
  showLoading();
  setLoadingProgress(0,1,'กำลังอ่านไฟล์...');
  const reader=new FileReader();
  reader.onload=async function(){
    pdfArrayBuffer=this.result.slice(0); // เก็บ copy ไว้ใช้ตอน export
    const pdf=await pdfjsLib.getDocument(new Uint8Array(this.result)).promise;
    pdfDoc=pdf; pageNum=1; pageRotations={}; pageDrawings={};
    currentZoom=1; canvasOffsetX=0; canvasOffsetY=0;
    Object.keys(pageImageCache).forEach(k=>delete pageImageCache[k]);
    MeasurementEngine.resetCalibration();
    VectorEngine.clearAll();
    selectedPages.clear();
    _initPageOrder(); // สร้าง pageOrder + pdfSources
    document.getElementById('thumb-placeholder').style.display='none';
    renderThumbnails();
    // ── Auto-detect V/R ทุกหน้า (background, lightweight) ──
    setTimeout(()=>autoDetectAllPages(), 500);

    // ── Priority: render 10 หน้าแรก ──
    const priority=Math.min(PRIORITY_COUNT, pdf.numPages);
    for(let i=1;i<=priority;i++){
      setLoadingProgress(i-1, priority, `render หน้า ${i}/${priority}...`);
      await renderPageToCache(pdf, i, 0);
      setLoadingProgress(i, priority, `หน้า ${i} เสร็จแล้ว`);
    }
    // fill thumbnails ที่ render แล้ว
    _fillThumbCanvases();

    // ── เปิดใช้งานได้เลย ──
    hideLoading();
    updateScaleDisplay(); renderPage(pageNum, 0);
    enableAllButtons(); disableMeasurementTools();

    // ── Background: render หน้าที่เหลือเงียบๆ ──
    if(pdf.numPages>priority){
      renderBackground(pdf, priority+1);
    }
  };
  reader.readAsArrayBuffer(file);
});

// ── pageTypes: เก็บผล detect ทุกหน้า ──
const pageTypes = {};  // {1:'vector', 2:'raster', ...}

// ── auto-detect ทุกหน้า lightweight หลัง PDF โหลด ──
async function autoDetectAllPages(){
  if(!pdfDoc) return;
  const total = pdfDoc.numPages;
  // รัน parallel batch ละ 4 หน้า — เร็วกว่า sequential มาก
  const BATCH = 4;
  for(let start=1; start<=total; start+=BATCH){
    const batch = [];
    for(let i=start; i<start+BATCH && i<=total; i++) batch.push(i);
    await Promise.all(batch.map(async i => {
      try {
        const page = await pdfDoc.getPage(i);
        const type = await VectorEngine.detectPDFType(page);
        pageTypes[i] = type;
      } catch(e){ pageTypes[i]='raster'; }
      _updateThumbVRBadge(i);
    }));
  }
}

function _updateThumbVRBadge(i){
  const wrap = document.getElementById(`thumb-${i}`);
  if(!wrap) return;
  let badge = wrap.querySelector('.vr-badge');
  if(!badge){ badge=document.createElement('span'); badge.className='vr-badge'; wrap.appendChild(badge); }
  const type = pageTypes[i];
  const cands = VectorEngine.getCandidates(i);
  if(!type){ badge.textContent=''; return; }
  if(VectorEngine.isExtracting(i)){
    badge.textContent='⏳'; badge.className='vr-badge vr-extracting';
  } else if(cands.length > 0){
    badge.textContent=`V ${cands.length}`; badge.className='vr-badge vr-vector-done';
  } else if(type==='vector'){
    badge.textContent='V'; badge.className='vr-badge vr-vector';
  } else {
    badge.textContent='R'; badge.className='vr-badge vr-raster';
  }
}

// force extract (Shift+click thumbnail)
async function _forceExtract(i){
  if(!pdfDoc) return;
  VectorEngine.clearPage(i);
  pageTypes[i] = 'vector';
  _updateThumbVRBadge(i);
  const page = await pdfDoc.getPage(i);
  const rot = (page.rotate + (pageRotations[i]||0)) % 360;
  updateStatus(`⏳ Force extract หน้า ${i}...`);
  await VectorEngine.extractSnapCandidates(page, rot);
  _updateThumbVRBadge(i);
  if(i===pageNum){ updateVectorBadge(i); updateStatus(`✅ Extract เสร็จ — ${VectorEngine.getCandidates(i).length} pts`); setTimeout(()=>handleStatusUpdate(),2000); }
}

  wrap.style.borderLeft=tag ? `3px solid ${colors[tag]||'#888'}` : '';
}

// PAGE RENDER  — ดึงจาก cache / render on-demand ถ้ายังไม่มี
// ──────────────────────────────────────────────
async function renderPage(num, srcId=0){
  const cacheKey = `${srcId}_${num}`;
  const src = pdfSources.find(s=>s.id===srcId) || {doc:pdfDoc, id:0};
  if(!pageImageCache[cacheKey]){
    updateStatus(`⏳ กำลัง render หน้า ${num}...`);
    await renderPageToCache(src.doc, num, srcId);
  }
  const dataURL=pageImageCache[cacheKey];
  if(!dataURL){ updateStatus('❌ โหลดหน้าไม่ได้'); return; }
  updateStatus('กำลังแสดงหน้า...');
  fabric.Image.fromURL(dataURL, function(img){
    baseCanvasWidth=img.width; baseCanvasHeight=img.height;
    canvas.setWidth(baseCanvasWidth*currentZoom);
    canvas.setHeight(baseCanvasHeight*currentZoom);
    canvas.setZoom(currentZoom); canvas.clear();
    canvas.setBackgroundImage(img, canvas.renderAll.bind(canvas),{originX:'left',originY:'top'});
    document.getElementById('page-info').textContent=`${num}/${pageOrder.length}`;
    _markActivePage();
    applyCanvasOffset();
    updateVectorBadge(num);
    if(pageDrawings[num]){
      canvas.loadFromJSON(JSON.parse(pageDrawings[num]),()=>{
        restorePairedLabels();
        canvas.renderAll(); runCheckEngine(); updateParkingCount();
      });
    } else { runCheckEngine(); updateParkingCount(); }
    updateStyle();
    updateStatus('🖐️ Pan — Scroll=Zoom | MiddleMouse/Space+Drag=Pan');
  });
}

function applyCanvasOffset(){
  const wrapper=document.getElementById('canvas-wrapper');
  wrapper.style.transform=`translate(${canvasOffsetX}px,${canvasOffsetY}px)`;
}

function saveCurrentPage(){ if(pdfDoc) pageDrawings[pageNum]=JSON.stringify(canvas.toJSON(['customData'])); }
function changePageTo(pos){
  if(!pdfDoc || !pageOrder[pos]) return;
  saveCurrentPage();
  const {srcId, page:pn} = pageOrder[pos];
  pageNum = pn;
  renderPage(pn, srcId);
}
function changePage(offset){ if(!pdfDoc)return; const n=pageNum+offset; if(n>=1&&n<=pdfDoc.numPages) changePageTo(n); }
async function rotatePage(){
  if(!pdfDoc)return;
  const objs=canvas.getObjects();
  if(objs.length>0){
    alert('ไม่สามารถหมุนได้ เนื่องจากมีการวาดแล้วในหน้านี้\nกรุณาลบ annotation ก่อนหมุน');
    return;
  }
  pageRotations[pageNum]=((pageRotations[pageNum]||0)+90)%360;
  const cacheKey=`0_${pageNum}`;
  delete pageImageCache[cacheKey];
  delete pageDrawings[pageNum];
  VectorEngine.clearPage(pageNum);
  await renderPageToCache(pdfDoc, pageNum, 0);
  renderPage(pageNum, 0);
}

// ZOOM  (scroll wheel + button)
// ──────────────────────────────────────────────
function changeZoom(delta, focalX, focalY){
  if(!pdfDoc) return;
  const oldZoom=currentZoom;
  currentZoom=Math.min(5,Math.max(0.1,currentZoom+delta));

  // focal point pan compensation
  if(focalX!==undefined && focalY!==undefined){
    const scale=currentZoom/oldZoom;
    canvasOffsetX=focalX-(focalX-canvasOffsetX)*scale;
    canvasOffsetY=focalY-(focalY-canvasOffsetY)*scale;
  }

  canvas.setWidth(baseCanvasWidth*currentZoom);
  canvas.setHeight(baseCanvasHeight*currentZoom);
  canvas.setZoom(currentZoom);
  document.getElementById('zoom-display').textContent=Math.round(currentZoom*100)+'%';
  applyCanvasOffset();
}

function resetZoom(){
  currentZoom=1; canvasOffsetX=0; canvasOffsetY=0;
  if(!pdfDoc) return;
  canvas.setWidth(baseCanvasWidth); canvas.setHeight(baseCanvasHeight);
  canvas.setZoom(1);
  document.getElementById('zoom-display').textContent='100%';
  applyCanvasOffset();
}

  if(!pdfDoc) return;
  e.preventDefault();
  const rect=this.getBoundingClientRect();
  const focalX=e.clientX-rect.left;
  const focalY=e.clientY-rect.top;
  const delta=e.deltaY>0 ? -0.15 : 0.15;
  changeZoom(delta, focalX, focalY);
},{ passive:false });

// PAN  (middle mouse + space+drag + 2-finger drag iPad)

let spaceDown=false, middleDown=false, shiftOrthoActive=false;

document.addEventListener('keydown', e=>{
  if(e.code==='Space'&&!e.repeat){ spaceDown=true; if(currentMode==='pan'||spaceDown) document.getElementById('canvas-scroll').style.cursor='grab'; }
  // Ctrl+Z = undo
  if((e.ctrlKey||e.metaKey) && e.key==='z' && !e.target.matches('input,textarea')){
    e.preventDefault(); undoLast(); return;
  }
  // Shift ค้าง = ortho ชั่วคราว (เฉพาะตอนวาด)
  if(e.code==='ShiftLeft'||e.code==='ShiftRight'){
    const drawModes=['polygon','measure-line','setback','calibrate'];
    if(drawModes.includes(currentMode) && !MeasurementEngine.getSnapFlags().ortho){
      shiftOrthoActive=true;
      MeasurementEngine.setSnapFlag('ortho', true);
    }
  }
  // DEL / Backspace — ลบ object ที่เลือก (ไม่ลบถ้ากำลังพิมพ์ข้อความ)
  if((e.key==='Delete'||e.key==='Backspace') && !e.target.matches('input,textarea')){
    const active=canvas.getActiveObjects();
    if(active.length){ e.preventDefault(); deleteSelected(); }
  }
});
document.addEventListener('keyup', e=>{
  if(e.code==='Space'){ spaceDown=false; document.getElementById('canvas-scroll').style.cursor=''; }
  if(e.code==='ShiftLeft'||e.code==='ShiftRight'){
    if(shiftOrthoActive){
      shiftOrthoActive=false;
      MeasurementEngine.setSnapFlag('ortho', false);
    }
  }
});

const scrollEl=document.getElementById('canvas-scroll');

scrollEl.addEventListener('mousedown',e=>{
  if(e.button===1||spaceDown){  // middle mouse or space
    e.preventDefault(); isPanning=true; panLastX=e.clientX; panLastY=e.clientY;
    scrollEl.style.cursor='grabbing';
  }
});
document.addEventListener('mousemove',e=>{
  if(!isPanning) return;
  canvasOffsetX+=e.clientX-panLastX;
  canvasOffsetY+=e.clientY-panLastY;
  panLastX=e.clientX; panLastY=e.clientY;
  applyCanvasOffset();
});
document.addEventListener('mouseup',e=>{
  if(e.button===1||isPanning){ isPanning=false; scrollEl.style.cursor=''; }
});

// Touch: 2-finger pan + pinch zoom
let touches={}, lastPinchDist=null, lastTouchMid=null;
scrollEl.addEventListener('touchstart',e=>{
  Array.from(e.changedTouches).forEach(t=>touches[t.identifier]={x:t.clientX,y:t.clientY});
  if(Object.keys(touches).length===2){
    const ts=Object.values(touches);
    lastPinchDist=Math.hypot(ts[1].x-ts[0].x,ts[1].y-ts[0].y);
    lastTouchMid={x:(ts[0].x+ts[1].x)/2,y:(ts[0].y+ts[1].y)/2};
  }
},{passive:false});

scrollEl.addEventListener('touchmove',e=>{
  e.preventDefault();
  Array.from(e.changedTouches).forEach(t=>touches[t.identifier]={x:t.clientX,y:t.clientY});
  const ts=Object.values(touches);
  if(ts.length===2){
    // pinch zoom
    const dist=Math.hypot(ts[1].x-ts[0].x,ts[1].y-ts[0].y);
    const mid={x:(ts[0].x+ts[1].x)/2,y:(ts[0].y+ts[1].y)/2};
    if(lastPinchDist){
      const rect=scrollEl.getBoundingClientRect();
      const delta=(dist-lastPinchDist)*0.01;
      changeZoom(delta,mid.x-rect.left,mid.y-rect.top);
    }
    // 2-finger pan
    if(lastTouchMid){
      canvasOffsetX+=mid.x-lastTouchMid.x;
      canvasOffsetY+=mid.y-lastTouchMid.y;
      applyCanvasOffset();
    }
    lastPinchDist=dist; lastTouchMid=mid;
  }
},{passive:false});

scrollEl.addEventListener('touchend',e=>{
  Array.from(e.changedTouches).forEach(t=>delete touches[t.identifier]);
  if(Object.keys(touches).length<2){ lastPinchDist=null; lastTouchMid=null; }
},{passive:false});

// ──────────────────────────────────────────────
// FABRIC PAN (mode=pan, 1 finger/left drag)
// ──────────────────────────────────────────────
let fabricPanning=false, fabricPanLast={x:0,y:0};
canvas.on('mouse:down',function(o){
  if(currentMode==='pan'){
    fabricPanning=true;
    fabricPanLast={x:o.e.clientX||o.e.touches?.[0]?.clientX,y:o.e.clientY||o.e.touches?.[0]?.clientY};
  }
});
canvas.on('mouse:move',function(o){
  // ghost line for measurement modes
  if(['setback','polygon','measure-line','calibrate'].includes(currentMode)){
    const pt=canvas.getPointer(o.e);
    MeasurementEngine.updateGhostLine(pt);
  }
  // fabric pan
  if(fabricPanning&&currentMode==='pan'){
    const cx=o.e.clientX||o.e.touches?.[0]?.clientX;
    const cy=o.e.clientY||o.e.touches?.[0]?.clientY;
    if(cx===undefined) return;
    canvasOffsetX+=cx-fabricPanLast.x;
    canvasOffsetY+=cy-fabricPanLast.y;
    fabricPanLast={x:cx,y:cy};
    applyCanvasOffset();
  }
});
canvas.on('mouse:up',function(){ fabricPanning=false; });

// ── Paired label: ขยับตาม object ──
canvas.on('mouse:down', function(e){
  const obj = canvas.getActiveObject();
  if(!obj||!obj.customData||!obj.customData.pairedLabel) return;
  obj._dragStartLeft = obj.left;
  obj._dragStartTop  = obj.top;
  obj._dragStartLblLeft = obj.customData.pairedLabel.left;
  obj._dragStartLblTop  = obj.customData.pairedLabel.top;
});
canvas.on('object:moving', function(e){
  const obj = e.target;
  if(!obj||!obj.customData||!obj.customData.pairedLabel) return;
  if(obj._dragStartLeft===undefined) return;
  const lbl = obj.customData.pairedLabel;
  const dx = obj.left - obj._dragStartLeft;
  const dy = obj.top  - obj._dragStartTop;
  lbl.set({ left: obj._dragStartLblLeft+dx, top: obj._dragStartLblTop+dy });
  lbl.setCoords();
  canvas.requestRenderAll();
});

async function renderAnnotationLayer(num, srcId=0){
  const cacheKey = `${srcId}_${num}`;
  const imgData=pageImageCache[cacheKey];
  if(!imgData) return null;
  return new Promise(resolve=>{
    fabric.Image.fromURL(imgData, function(bgImg){
      const w=bgImg.width, h=bgImg.height;
      const el=document.createElement('canvas');
      el.width=w; el.height=h;
      const tempCanvas=new fabric.Canvas(el,{width:w,height:h,enableRetinaScaling:false});
      tempCanvas.setBackgroundImage(bgImg, ()=>{});
      const drawingData=pageDrawings[num];
      if(drawingData && JSON.parse(drawingData).objects?.length>0){
        tempCanvas.loadFromJSON(JSON.parse(drawingData),()=>{
          tempCanvas.backgroundImage=bgImg;
          tempCanvas.renderAll();
          const dataURL=tempCanvas.toDataURL({format:'jpeg',quality:0.92,multiplier:1});
          tempCanvas.dispose();
          resolve(dataURL);
        });
      } else {
        // ไม่มี annotation — ใช้ JPEG เดิม
        resolve(imgData);
      }
    });
  });
}


function restorePairedLabels(){
  // หลัง loadFromJSON — pairedLabel reference หาย ต้อง re-link
  // แต่ละ polygon/line จะมี label อยู่ถัดไปใน objects array
  const objs = canvas.getObjects();
  for(let i=0;i<objs.length-1;i++){
    const obj=objs[i];
    const next=objs[i+1];
    if(!obj.customData) continue;
    if((obj.customData.isPolygon||obj.customData.isMeasure||obj.customData.isSetback) && next.customData&&next.customData.isPairedLabel){
      obj.customData.pairedLabel = next;
    }
  }
}

// ── Hatch pattern สำหรับ zone ──────────────────────
function _makeHatchPattern(color, spacing=12, opacity=0.45){
  // SVG diagonal lines pattern
  const c = color || '#7c3aed';
  const svg = `<svg xmlns='http://www.w3.org/2000/svg' width='${spacing}' height='${spacing}'>
    <line x1='0' y1='${spacing}' x2='${spacing}' y2='0' stroke='${c}' stroke-width='1.5' stroke-opacity='${opacity}'/>
  </svg>`;
  const url = 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
  return new Promise(resolve => {
    fabric.util.loadImage(url, img => {
      resolve(new fabric.Pattern({ source: img, repeat: 'repeat' }));
    });
  });
}

// zone preview rect (ยังไม่ใช่ pattern เพราะ async)
let _zonePreview = null;

async function _finalizeZone(x, y, w, h, color){
  if(w < 5 || h < 5) return;
  const pattern = await _makeHatchPattern(color);
  const zone = new fabric.Rect({
    left: x, top: y, width: w, height: h,
    fill: pattern,
    stroke: color, strokeWidth: 1.5 / currentZoom,
    opacity: 1,
    selectable: true, evented: true,
    customData: { isZone: true, color }
  });
  // label กลาง zone
  const label = new fabric.IText('', {
    left: x + w/2, top: y + h/2,
    fontSize: Math.max(12, Math.min(w, h) * 0.08),
    fill: color, fontFamily: 'sans-serif',
    originX: 'center', originY: 'center',
    textAlign: 'center', selectable: true,
    customData: { isZoneLabel: true, pairedZone: zone }
  });
  zone.customData.pairedLabel = label;
  canvas.add(zone, label);
  // double-click zone → edit label
  zone.on('mousedblclick', () => {
    canvas.setActiveObject(label);
    label.enterEditing();
    label.selectAll();
    canvas.renderAll();
  });
  canvas.renderAll();
  saveCurrentPage();
  setMode('select');
}

canvas.on('mouse:down',function(o){
  if(['none','pan','select'].includes(currentMode)||canvas.isDrawingMode) return;
  if(fabricPanning) return;
  const pt=canvas.getPointer(o.e);
  const color=document.getElementById('tool-color').value;
  const sw2=parseInt(document.getElementById('tool-width').value);

  if(currentMode==='text'){
    const txt=new fabric.IText('คลิกพิมพ์...',{left:pt.x,top:pt.y,fontFamily:'sans-serif',fill:color,fontSize:18*(sw2/2)});
    canvas.add(txt); canvas.setActiveObject(txt); txt.enterEditing(); txt.selectAll(); setMode('select'); return;
  }
  if(currentMode==='rect'||currentMode==='circle'||currentMode==='zone'){
    isDrawingShape=true; origX=pt.x; origY=pt.y;
    if(currentMode==='zone'){
      // preview rect โปร่งใส ก่อน pattern จะ load
      activeShape=new fabric.Rect({left:origX,top:origY,width:0,height:0,
        fill:'rgba(124,58,237,0.15)',stroke:color||'#7c3aed',
        strokeWidth:1.5/currentZoom,strokeDashArray:[4/currentZoom,3/currentZoom],
        selectable:false});
    } else {
      activeShape=currentMode==='rect'
        ?new fabric.Rect({left:origX,top:origY,width:0,height:0,fill:'transparent',stroke:color,strokeWidth:sw2/currentZoom,selectable:false})
        :new fabric.Ellipse({left:origX,top:origY,rx:0,ry:0,fill:'transparent',stroke:color,strokeWidth:sw2/currentZoom,selectable:false});
    }
    canvas.add(activeShape); return;
  }
  if(currentMode==='calibrate'){    MeasurementEngine.handleCalibrate(pt);    handleStatusUpdate(); return; }
  if(currentMode==='measure-line'){ MeasurementEngine.handleMeasureLine(pt);  handleStatusUpdate(); return; }
  if(currentMode==='polygon'){
    // ── Auto-close: ถ้า ≥3 จุดแล้ว และคลิกใกล้จุดแรก → ปิด polygon อัตโนมัติ
    if(MeasurementEngine.getTempPointCount() >= 2){
      const firstPt = MeasurementEngine.getFirstPoint();
      if(firstPt){
        const closeTol = MeasurementEngine.getSnapThreshold().pt / currentZoom;
        if(Math.hypot(pt.x - firstPt.x, pt.y - firstPt.y) < closeTol){
          // trigger finish
          const labels={site:'ที่ดิน',building:'อาคาร',space:'ที่ว่าง',floor:'พื้นที่ใช้สอย',parking:'ที่จอดรถ',stair:'บันได',roof:'ดาดฟ้า'};
          const label = currentPolyType.startsWith('custom_') ? currentPolyType.replace('custom_','') : (labels[currentPolyType]||currentPolyType);
          MeasurementEngine.finishPolygon(currentPolyType, label);
          const objs=canvas.getObjects('polygon');
          if(objs.length>0){ const last=objs[objs.length-1]; if(last.customData) last.customData.pageLabel=pageLabels[pageNum]||''; }
          setMode('select'); saveCurrentPage(); handleStatusUpdate();
          return;
        }
      }
    }
    MeasurementEngine.handlePolygonClick(pt,currentPolyType); handleStatusUpdate(); return;
  }
  if(currentMode==='setback'){      MeasurementEngine.handleSetback(pt);       handleStatusUpdate(); return; }
  if(currentMode==='park-normal'||currentMode==='park-disabled'){ handleParkingPin(pt,currentMode); return; }
});
canvas.on('mouse:move',function(o){
  if(!isDrawingShape) return;
  const pt=canvas.getPointer(o.e);
  if(currentMode==='rect'||currentMode==='zone') activeShape.set({width:Math.abs(origX-pt.x),height:Math.abs(origY-pt.y),left:Math.min(pt.x,origX),top:Math.min(pt.y,origY)});
  else if(currentMode==='circle') activeShape.set({rx:Math.abs(origX-pt.x)/2,ry:Math.abs(origY-pt.y)/2,left:Math.min(pt.x,origX),top:Math.min(pt.y,origY)});
  canvas.renderAll();
});
canvas.on('mouse:up',function(){
  if(!isDrawingShape) return;
  isDrawingShape=false;
  if(currentMode==='zone' && activeShape){
    const x=activeShape.left, y=activeShape.top;
    const w=activeShape.width, h=activeShape.height;
    const color=document.getElementById('tool-color').value||'#7c3aed';
    canvas.remove(activeShape); activeShape=null;
    _finalizeZone(x, y, w, h, color);
  } else {
    activeShape&&activeShape.setCoords(); activeShape=null; setMode('select');
  }
});

function updateVectorBadge(num){
  const el = document.getElementById('vector-badge');
  if (!el) return;
  const type = VectorEngine.getType(num);
  if (!type) { el.style.display = 'none'; return; }
  if (type === 'vector') {
    const count = VectorEngine.getCandidates(num).length;
    if (count > 0) {
      el.textContent = `🟢 Vector snap (${count} pts)`;
      el.style.borderColor = 'rgba(48,209,88,0.4)';
    } else {
      // กำลัง extract อยู่
      el.textContent = `⏳ กำลังโหลด snap...`;
      el.style.borderColor = 'rgba(255,214,10,0.4)';
    }
  } else {
    el.textContent = '🟡 Scan — ไม่มี snap เส้น';
    el.style.borderColor = 'rgba(255,214,10,0.4)';
  }
  el.style.display = 'block';
}

// ── trigger extract สำหรับหน้าที่ระบุ ──
async function _triggerExtract(i) {
  if (!pdfDoc) return;
  const page = await pdfDoc.getPage(i);
  const rot  = (page.rotate + (pageRotations[i] || 0)) % 360;

  const type = await VectorEngine.detectPDFType(page);
  pageTypes[i] = type;
  _updateThumbVRBadge(i);
  if (i === pageNum) updateVectorBadge(i);

  if (type !== 'vector') return;

  updateStatus(`⏳ กำลังโหลด snap points หน้า ${i}...`);
  _updateThumbVRBadge(i); // แสดง ⏳
  await VectorEngine.extractSnapCandidates(page, rot);
  _updateThumbVRBadge(i); // แสดง V + count

  if (i === pageNum) {
    updateVectorBadge(i);
    updateStatus('✅ Snap พร้อมใช้งาน — คลิกบนเส้น PDF เพื่อ snap');
    setTimeout(() => handleStatusUpdate(), 2500);
  }
}
function handleStatusUpdate(){
  const n=MeasurementEngine.getTempPointCount();

  // ── hint: ถ้าอยู่ใน measure mode แต่หน้ายังไม่ tag ──
  const measureModes = ['measure-line','polygon','setback','calibrate'];
  if (measureModes.includes(currentMode)) {
    const tag = pageLabels[pageNum] || '';
    const hasSnap = VectorEngine.getCandidates(pageNum).length > 0;
    if (!tag || tag === 'other') {
      // แสดง hint แทรกท้าย status (จะถูก override ด้วย msg ปกติ ถ้ามี)
    }
  }

  const msgs={
    calibrate: ['🎯 คลิกจุดที่ 1 บนเส้นอ้างอิง','🎯 คลิกจุดที่ 2 → กรอกระยะจริง'],
    'measure-line': ['📏 คลิกจุดเริ่มต้น','📏 คลิกจุดสิ้นสุด'],
    setback: ['🛣️ คลิกจุดที่ 1 บนเส้นถนน (ชมพู)','🛣️ คลิกจุดที่ 2 บนเส้นถนน (ม่วง)','🛣️ คลิกจุดบนแนวอาคาร (ส้ม)'],
  };

  // สร้าง suffix hint snap
  const snapHint = (() => {
    if (!pdfDoc) return '';
    const tag = pageLabels[pageNum] || '';
    const hasSnap = VectorEngine.getCandidates(pageNum).length > 0;
    if (hasSnap) return '';  // snap พร้อมแล้ว ไม่ต้อง hint
    if (!tag || tag === 'other') return '  💡 ระบุประเภทหน้าเพื่อเปิด snap เส้น PDF';
    return '';  // tag แล้วแต่กำลัง extract (มี badge แสดงอยู่แล้ว)
  })();

  if(msgs[currentMode]) { updateStatus(msgs[currentMode][Math.min(n,msgs[currentMode].length-1)] + snapHint); return; }
  if(currentMode==='polygon'){
    const typeLabels={site:'ที่ดิน',building:'อาคาร',space:'ที่ว่าง',floor:'พื้นที่ใช้สอย',parking:'ที่จอดรถ',stair:'บันได',roof:'ดาดฟ้า'};
    const typeLabel=currentPolyType.startsWith('custom_')?currentPolyType.replace('custom_',''):(typeLabels[currentPolyType]||currentPolyType);
    const closeHint = n >= 3 ? ' — คลิกจุดแรก 🟢 เพื่อปิด' : '';
    updateStatus(`📐 วัดพื้นที่ (${typeLabel}): [${n} จุด]${closeHint}` + snapHint); return;
  }
  if(currentMode==='park-normal')   { updateStatus('🚗 คลิกบนแบบเพื่อนับที่จอดรถปกติ'); return; }
  if(currentMode==='park-disabled') { updateStatus('♿ คลิกบนแบบเพื่อนับที่จอดรถคนพิการ'); return; }
  if(currentMode==='zone')   { updateStatus('🟪 ลาก drag เพื่อสร้าง Zone ไฮไลท์ — ดับเบิลคลิกเพื่อพิมพ์ label'); return; }
  if(currentMode==='pan')    { updateStatus('🖐️ Pan — Scroll=Zoom, MiddleMouse/Space+Drag=Pan, Pinch=Zoom(iPad)'); return; }
  if(currentMode==='select') { updateStatus('↖️ เลือก/ย้าย'); return; }
  updateStatus('โหมด: '+currentMode);
}

function toggleSnap(key){
  const flags = MeasurementEngine.getSnapFlags();
  const newVal = !flags[key];
  MeasurementEngine.setSnapFlag(key, newVal);
  const chk = document.getElementById('snapchk-'+key);
  if(chk) chk.classList.toggle('off', !newVal);
  // อัพเดต 🧲 button — dim ถ้า snap ทั้งหมดปิด
  const allOff = !Object.values(MeasurementEngine.getSnapFlags()).some(v=>v);
  const btn = document.getElementById('btn-snap');
  if(btn) btn.style.opacity = allOff ? '0.4' : '1';
}

function setMode(mode){
  currentMode=mode; MeasurementEngine.clearTemps();
  canvas.isDrawingMode=(mode==='pen'||mode==='highlight');
  updateStyle();
  document.querySelectorAll('#top-bar .tb-btn, #ipad-bottom-bar .ipad-tool-btn').forEach(b=>b.classList.remove('active','btn-active'));
  if(document.getElementById('btn-'+mode)) document.getElementById('btn-'+mode).classList.add('active');
  if(document.getElementById('ipad-'+mode)) document.getElementById('ipad-'+mode).classList.add('active');
  document.getElementById('btn-finish').classList.toggle('hide',mode!=='polygon');
  if(mode==='select'){ canvas.selection=true; canvas.defaultCursor='default'; canvas.forEachObject(o=>o.set('selectable',true)); }
  else { canvas.selection=false; canvas.defaultCursor=mode==='pan'?'grab':'crosshair'; canvas.discardActiveObject(); canvas.forEachObject(o=>o.set('selectable',false)); }
  canvas.requestRenderAll(); handleStatusUpdate(); closeDropdowns();
}
function handleParkingPin(pt, mode){
  const isDisabled = mode==='park-disabled';
  const color = isDisabled ? '#28a745' : '#0056b3';
  const icon  = isDisabled ? '♿' : '🚗';
  const r = 12/currentZoom;
  const circle = new fabric.Circle({
    left:pt.x-r, top:pt.y-r, radius:r,
    fill:color, stroke:'#fff', strokeWidth:2/currentZoom,
    selectable:true, evented:true,
    customData:{ isParking:true, parkType: isDisabled?'disabled':'normal' }
  });
  const label = new fabric.Text(icon, {
    left:pt.x, top:pt.y-r*2.5,
    fontSize:Math.max(10, 14/currentZoom),
    originX:'center', selectable:false, evented:false,
  });
  canvas.add(circle, label);
  canvas.renderAll();
  saveCurrentPage();
  updateParkingCount();
}

function updateParkingCount(){
  let normal=0, disabled=0;
  canvas.getObjects().forEach(o=>{
    if(o.customData&&o.customData.isParking){
      if(o.customData.parkType==='disabled') disabled++;
      else normal++;
    }
  });
  document.getElementById('panel-park-normal').textContent=normal;
  document.getElementById('panel-park-disabled').textContent=disabled;
  document.getElementById('panel-park-total').textContent=normal+disabled;
  document.getElementById('park-normal-count').textContent=normal;
  document.getElementById('park-disabled-count').textContent=disabled;
  const counter=document.getElementById('park-counter');
  counter.style.display=(normal+disabled>0)?'block':'none';
}

function undoLast(){
  const objs=canvas.getObjects();
  if(!objs.length) return;
  const last=objs[objs.length-1];
  // ถ้า last เป็น label ที่มี pairedZone → ลบ zone ด้วย
  if(last.customData?.isZoneLabel && last.customData.pairedZone){
    canvas.remove(last.customData.pairedZone);
  }
  // ถ้า last เป็น zone/polygon ที่มี pairedLabel → ลบ label ด้วย
  if(last.customData?.pairedLabel){
    canvas.remove(last.customData.pairedLabel);
  }
  canvas.remove(last);
  canvas.renderAll();
  saveCurrentPage(); runCheckEngine();
}

function showContextBar(obj){
  const bar=document.getElementById('context-bar');
  const lbl=document.getElementById('ctx-label');
  const colorInput=document.getElementById('ctx-color');
  // label
  let info='วัตถุ';
  if(obj.customData){
    if(obj.customData.isPolygon){ const t={site:'ที่ดิน',building:'อาคาร',space:'ที่ว่าง',floor:'พื้นที่',parking:'จอดรถ',stair:'บันได',roof:'ดาดฟ้า'}; info=(t[obj.customData.type]||obj.customData.type||'พื้นที่')+` — ${(obj.customData.area||0).toFixed(2)} ตร.ม.`; }
    else if(obj.customData.isMeasure){ info=`ระยะ ${(obj.customData.val||0).toFixed(2)} ม.`; }
    else if(obj.customData.isSetback){ info=`ระยะร่น ${(obj.customData.val||0).toFixed(2)} ม.`; }
    else if(obj.customData.isPairedLabel){ info='ป้ายชื่อ'; }
  }
  lbl.textContent=info;
  // สีปัจจุบัน
  const c=obj.fill||obj.stroke||'#ffffff';
  if(typeof c==='string'&&c.startsWith('#')) colorInput.value=c;
  // position — ลอยเหนือ object
  const canvasEl=document.getElementById('drawing-canvas');
  const rect=canvasEl.getBoundingClientRect();
  const objBound=obj.getBoundingRect();
  const x=rect.left + (objBound.left+objBound.width/2)*currentZoom + window.scrollX;
  const y=rect.top  + objBound.top*currentZoom + window.scrollY - 52;
  bar.style.left=Math.max(8, Math.min(x-160, window.innerWidth-340))+'px';
  bar.style.top=Math.max(56, y)+'px';
  bar.classList.add('visible');
}

function applyContextColor(color){
  const objs=canvas.getActiveObjects();
  objs.forEach(o=>{
    if(o.fill&&o.fill!=='transparent') o.set('fill', hexToRgba(color,0.4));
    if(o.stroke) o.set('stroke', color);
  });
  canvas.requestRenderAll(); saveCurrentPage();
}
function hexToRgba(hex,alpha){
  const r=parseInt(hex.slice(1,3),16),g=parseInt(hex.slice(3,5),16),b=parseInt(hex.slice(5,7),16);
  return `rgba(${r},${g},${b},${alpha})`;
}

// Floating panel drag
(function(){
  const panel=document.getElementById('float-panel');
  const header=document.getElementById('float-panel-header');
  let dragging=false,ox=0,oy=0,startRight=0,startTop=0;
  header.addEventListener('mousedown',e=>{
    if(e.target.closest('button')) return;
    dragging=true;
    const rect=panel.getBoundingClientRect();
    ox=e.clientX; oy=e.clientY;
    startRight=window.innerWidth-rect.right;
    startTop=rect.top;
    panel.style.transition='none';
  });
  document.addEventListener('mousemove',e=>{
    if(!dragging) return;
    const dx=e.clientX-ox, dy=e.clientY-oy;
    panel.style.right=(startRight-dx)+'px';
    panel.style.top=Math.max(0,startTop+dy)+'px';
  });
  document.addEventListener('mouseup',()=>{ dragging=false; panel.style.transition=''; });
})();

function _initPageOrder(){
  pageOrder = [];
  pdfSources = [{id:0, buffer: pdfArrayBuffer, doc: pdfDoc}];
  for(let i=1; i<=pdfDoc.numPages; i++) pageOrder.push({srcId:0, page:i});

function _fillThumbCanvases(){
  pageOrder.forEach((entry, pos) => {
    const {srcId, page:pn} = entry;
    const uid = `${srcId}_${pn}`;
    const cacheKey = uid;
    const tc = document.getElementById(`thumb-canvas-${uid}`);
    if(!tc) return;
    const src = pdfSources.find(s=>s.id===srcId);
    if(!src) return;
    if(pageImageCache[cacheKey]){
      const img = new Image();
      img.onload = () => { tc.width=img.naturalWidth; tc.height=img.naturalHeight; tc.getContext('2d').drawImage(img,0,0); };
      img.src = pageImageCache[cacheKey];
    } else {
      // render thumbnail on-demand
      src.doc.getPage(pn).then(page => {
        const vp = page.getViewport({scale:0.25, rotation:(page.rotate+(pageRotations[pn]||0))%360});
        tc.width=Math.round(vp.width); tc.height=Math.round(vp.height);
        page.render({canvasContext:tc.getContext('2d'), viewport:vp}).promise;
      }).catch(()=>{});
    }
  });
}

// ── Drag reorder ────────────────────────────────
function _initThumbDrag(wrap, pos){
  wrap.setAttribute('draggable','true');
  wrap.addEventListener('dragstart', e => {
    _dragSrcPos = parseInt(wrap.dataset.pos);
    wrap.classList.add('dragging-page');
    e.dataTransfer.effectAllowed='move';
  });
  wrap.addEventListener('dragend', () => {
    wrap.classList.remove('dragging-page');
    document.querySelectorAll('.thumb-wrap').forEach(w=>w.classList.remove('drag-over-top','drag-over-bottom'));
  });
  wrap.addEventListener('dragover', e => {
    e.preventDefault();
    const tPos = parseInt(wrap.dataset.pos);
    document.querySelectorAll('.thumb-wrap').forEach(w=>w.classList.remove('drag-over-top','drag-over-bottom'));
    wrap.classList.add(tPos < _dragSrcPos ? 'drag-over-top' : 'drag-over-bottom');
  });
  wrap.addEventListener('drop', e => {
    e.preventDefault();
    const tPos = parseInt(wrap.dataset.pos);
    if(_dragSrcPos < 0 || _dragSrcPos===tPos) return;
    const [moved] = pageOrder.splice(_dragSrcPos, 1);
    const insertAt = tPos > _dragSrcPos ? tPos : tPos;
    pageOrder.splice(insertAt, 0, moved);
    _dragSrcPos = -1;
    renderThumbnails();
    // ถ้า currentPos เปลี่ยน ให้แสดงหน้าใหม่
    const newPos = pageOrder.findIndex(e=>e.srcId===0 && e.page===pageNum);
    if(newPos>=0) currentDisplayPos = newPos;
    _markActivePage();
  });
}

function deleteSelectedPages(){
  if(selectedPages.size===0){ alert('เลือกหน้าที่ต้องการลบก่อน'); return; }
  if(pageOrder.length - selectedPages.size < 1){ alert('ไม่สามารถลบได้ — ต้องมีอย่างน้อย 1 หน้า'); return; }
  if(!confirm(`ลบ ${selectedPages.size} หน้า?`)) return;
  pageOrder = pageOrder.filter(e => !selectedPages.has(`${e.srcId}_${e.page}`));
  selectedPages.clear();
  renderThumbnails();
  // ถ้าหน้าปัจจุบันถูกลบ → ไปหน้าแรก
  const pos = pageOrder.findIndex(e=>e.srcId===0&&e.page===pageNum);
  changePageTo(pos>=0 ? pos : 0);
}

function deletePageAt(pos){
  if(pageOrder.length<=1){ alert('ต้องมีอย่างน้อย 1 หน้า'); return; }
  pageOrder.splice(pos, 1);
  renderThumbnails();
  const newPos = Math.min(pos, pageOrder.length-1);
  changePageTo(newPos);
}

// ── Duplicate pages ─────────────────────────────
function duplicateSelectedPages(){
  if(selectedPages.size===0){ alert('เลือกหน้าที่ต้องการทำซ้ำก่อน'); return; }
  const inserts = [];
  pageOrder.forEach((entry, pos) => {
    if(selectedPages.has(`${entry.srcId}_${entry.page}`))
      inserts.push({pos, entry: {...entry}});
  });
  // insert จากท้ายเพื่อไม่ให้ index เลื่อน
  inserts.reverse().forEach(({pos, entry}) => pageOrder.splice(pos+1, 0, entry));
  selectedPages.clear();
  renderThumbnails();
}

function duplicatePageAt(pos){
  const entry = pageOrder[pos];
  if(!entry) return;
  pageOrder.splice(pos+1, 0, {...entry});
  renderThumbnails();
}

  if(!file) return;

function _getExportPositions(scope){
  if(scope==='all') return pageOrder.map((_,i)=>i);
  if(scope==='current'){
    const pos = pageOrder.findIndex(e=>e.srcId===0&&e.page===pageNum);
    return pos>=0 ? [pos] : [0];
  }
  if(scope==='selected'){
    const positions=[];
    pageOrder.forEach((e,i)=>{ if(selectedPages.has(`${e.srcId}_${e.page}`)) positions.push(i); });
    return positions.length>0 ? positions : _getExportPositions('current');
  }
  return [];
}

async function exportPages(scope){
  saveCurrentPage();
  const positions = _getExportPositions(scope);
  if(!positions.length){ alert('ไม่มีหน้าที่เลือก'); return; }
  await _exportByPos(positions);
}

async function _exportByPos(positions){
  if(typeof PDFLib==='undefined'){ alert('โหลด pdf-lib ไม่สำเร็จ'); return; }
  updateStatus(`⏳ กำลัง export ${positions.length} หน้า...`);
  const {PDFDocument} = PDFLib;
  const newDoc = await PDFDocument.create();
  for(const pos of positions){
    const {srcId, page:pn} = pageOrder[pos];
    const cacheKey = `${srcId}_${pn}`;
    const src = pdfSources.find(s=>s.id===srcId);
    if(!src) continue;
    if(!pageImageCache[cacheKey]) await renderPageToCache(src.doc, pn, srcId);
    const dataURL = srcId===0 ? await renderAnnotationLayer(pn) : pageImageCache[cacheKey];
    if(!dataURL) continue;
    const base64 = dataURL.split(',')[1];
    const binary = atob(base64);
    const bytes = new Uint8Array(binary.length);
    for(let b=0;b<binary.length;b++) bytes[b]=binary.charCodeAt(b);
    const jpgImg = await newDoc.embedJpg(bytes);
    const {width,height} = jpgImg.scale(1);
    const pg = newDoc.addPage([width,height]);
    pg.drawImage(jpgImg,{x:0,y:0,width,height});
  }
  const pdfBytes = await newDoc.save();
  const blob = new Blob([pdfBytes],{type:'application/pdf'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href=url; a.download=`${pdfFileName}_export.pdf`; a.click();
  URL.revokeObjectURL(url);
  updateStatus(`✅ Export สำเร็จ — ${positions.length} หน้า`);
}

async function _printByPos(positions){
  if(typeof PDFLib==='undefined'){ alert('โหลด pdf-lib ไม่สำเร็จ'); return; }
  updateStatus(`⏳ กำลังเตรียมปริ้น ${positions.length} หน้า...`);
  const {PDFDocument} = PDFLib;
  const newDoc = await PDFDocument.create();
  for(const pos of positions){
    const {srcId, page:pn} = pageOrder[pos];
    const cacheKey = `${srcId}_${pn}`;
    const src = pdfSources.find(s=>s.id===srcId);
    if(!src) continue;
    if(!pageImageCache[cacheKey]) await renderPageToCache(src.doc, pn, srcId);
    const dataURL = srcId===0 ? await renderAnnotationLayer(pn) : pageImageCache[cacheKey];
    if(!dataURL) continue;
    const base64=dataURL.split(',')[1];
    const binary=atob(base64);
    const bytes=new Uint8Array(binary.length);
    for(let b=0;b<binary.length;b++) bytes[b]=binary.charCodeAt(b);
    const jpgImg=await newDoc.embedJpg(bytes);
    const {width,height}=jpgImg.scale(1);
    const pg=newDoc.addPage([width,height]);
    pg.drawImage(jpgImg,{x:0,y:0,width,height});
  }
  const pdfBytes=await newDoc.save();
  const blob=new Blob([pdfBytes],{type:'application/pdf'});
  const url=URL.createObjectURL(blob);
  // print via hidden iframe
  const iframe=document.createElement('iframe');
  iframe.style.cssText='position:fixed;top:-9999px;left:-9999px;width:0;height:0;border:none;';
  iframe.src=url;
  document.body.appendChild(iframe);
  iframe.onload=()=>{
    setTimeout(()=>{
      iframe.contentWindow.print();
      setTimeout(()=>{ document.body.removeChild(iframe); URL.revokeObjectURL(url); },2000);
    },500);
  };
  updateStatus(`✅ เปิด print dialog — ${positions.length} หน้า`);
}

// ── Resize handle drag ──
(function(){
  const handle = document.getElementById('thumb-resize-handle');
  const strip  = document.getElementById('thumb-strip');
  if(!handle || !strip) return;

  let dragging = false, startX = 0, startW = 0;

  handle.addEventListener('mousedown', e => {
    e.preventDefault();
    dragging = true;
    startX = e.clientX;
    startW = strip.offsetWidth;
    handle.classList.add('dragging');
    document.body.style.cursor = 'col-resize';
    document.body.style.userSelect = 'none';
  });

  document.addEventListener('mousemove', e => {
    if(!dragging) return;
    const delta = e.clientX - startX;
    const newW  = Math.max(80, Math.min(320, startW + delta));
    strip.style.width = newW + 'px';
    _thumbWidth = newW;
    // อัพเดต inner width ให้ thumbnail ปรับตาม
    const inner = document.getElementById('thumb-strip-inner');
    if(inner) inner.style.minWidth = Math.max(70, newW - 16) + 'px';
  });

  document.addEventListener('mouseup', () => {
    if(!dragging) return;
    dragging = false;
    handle.classList.remove('dragging');
    document.body.style.cursor = '';
    document.body.style.userSelect = '';
  });

  // touch support (iPad)
  handle.addEventListener('touchstart', e => {
    startX = e.touches[0].clientX;
    startW = strip.offsetWidth;
    dragging = true;
  }, {passive:true});
  document.addEventListener('touchmove', e => {
    if(!dragging) return;
    const delta = e.touches[0].clientX - startX;
    const newW  = Math.max(80, Math.min(320, startW + delta));
    strip.style.width = newW + 'px';
    _thumbWidth = newW;
  }, {passive:true});
  document.addEventListener('touchend', () => { dragging = false; });
})();

function runCheckEngine(){
  if(!MeasurementEngine.isCalibrated()) return;
  const pi=appState.projectInfo;
  const reqSB=pi.roadWidth>=20?6:pi.roadWidth>=10?5:3;
  const maxBcr=pi.buildingType==='residential'?50:70;
  const maxFar=pi.buildingType==='residential'?2.0:5.0;
  let siteArea=0,bldgArea=0,spaceArea=0,actSB=Infinity;
  appState.checkResults=[];
  const allObj=[...canvas.getObjects()];
  Object.keys(pageDrawings).forEach(p=>{ if(parseInt(p)!==pageNum&&pageDrawings[p]){ const parsed=JSON.parse(pageDrawings[p]); if(parsed.objects) allObj.push(...parsed.objects); } });
  allObj.forEach(obj=>{ if(!obj.customData)return; if(obj.customData.isPolygon){ if(obj.customData.type==='site')siteArea+=obj.customData.area; if(obj.customData.type==='building')bldgArea+=obj.customData.area; if(obj.customData.type==='space')spaceArea+=obj.customData.area; } if(obj.customData.isSetback) actSB=Math.min(actSB,obj.customData.val); });
  let html='';
  if(actSB!==Infinity){ const pass=actSB>=reqSB; const text=`${pass?'✅':'❌'} ระยะร่นถนน: ${actSB.toFixed(2)}ม. (≥${reqSB}ม.)`; appState.checkResults.push(text); html+=`<div class="check-item ${pass?'check-pass':'check-fail'}"><b>${text}</b></div>`; }
  else html+='<div class="check-item check-wait">⏳ ระยะร่นจากถนน: ยังไม่ได้วัด</div>';
  if(siteArea>0&&bldgArea>0){
    const bcr=(bldgArea/siteArea)*100, bcrP=bcr<=maxBcr;
    let bt=`${bcrP?'✅':'❌'} BCR: ${bcr.toFixed(1)}% (≤${maxBcr}%)`; if(!bcrP) bt+=`<br><small style="color:#dc3545">ลดพื้นที่ ${(bldgArea-siteArea*maxBcr/100).toFixed(2)} ตร.ม.</small>`;
    appState.checkResults.push(bt); html+=`<div class="check-item ${bcrP?'check-pass':'check-fail'}"><b>${bt}</b></div>`;
    const far=(bldgArea*pi.floors)/siteArea, farP=far<=maxFar;
    const ft=`${farP?'✅':'❌'} FAR: ${far.toFixed(2)} (≤${maxFar})`; appState.checkResults.push(ft); html+=`<div class="check-item ${farP?'check-pass':'check-fail'}"><b>${ft}</b></div>`;
  } else html+='<div class="check-item check-wait">⏳ BCR & FAR: รอวัดที่ดิน+อาคาร</div>';
  if(siteArea>0){ const sp=spaceArea>0?spaceArea:siteArea-bldgArea, sr=(sp/siteArea)*100, spP=sr>=30; const st=`${spP?'✅':'❌'} ที่ว่าง: ${sr.toFixed(1)}% (≥30%)`; appState.checkResults.push(st); html+=`<div class="check-item ${spP?'check-pass':'check-fail'}"><b>${st}</b></div>`; }
  else html+='<div class="check-item check-wait">⏳ ที่ว่าง: รอขอบเขตที่ดิน</div>';
  document.getElementById('checklist-auto-container').innerHTML=html;
}
